def calcularValorTrianguloPascal(fila: Int, columna: Int,
    metodoFactorial: Int => Long): Long = {
  if (columna > fila) return 0
  metodoFactorial(fila) /
    (metodoFactorial(columna) * metodoFactorial(fila - columna))
}

// tail-recursive factorial
def factorialTailRecursive(n: Int, a: Int): Int =
  if (n == 0) a
  else factorialTailRecursive(n - 1, n * a)
